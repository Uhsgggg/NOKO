**次にcppを作成**

`Source/Game/UI/HudPresenter.cpp`（ファイル全文・省略なし）

```cpp
#include "HudPresenter.h"
#include <cmath>

static float LimitRatio(float ratio)
{
    if (!std::isfinite(ratio) || ratio < 0)
        return 0;
    if (ratio > 1)
        return 1;
    return ratio;
}

HudView HudPresenter::Build(const HudInput& input, float maxDrag)
{
    HudView view;
    int shots = input.remainingShots;
    int enemies = input.nextEnemyCount;
    if (shots < 0) shots = 0;
    if (enemies < 0) enemies = 0;
    view.shotsText = std::to_string(shots);
    view.nextEnemiesText = std::to_string(enemies);

    if (input.bossMaxHP > 0)
    {
        float ratio = float(input.bossHP) / input.bossMaxHP;
        view.bossRatio = LimitRatio(ratio);
    }

    for (float hp : input.enemyHPRatios)
    {
        float ratio = LimitRatio(hp);
        view.enemyRatios.push_back(ratio);
        HudColor color;
        if (ratio > 0.5f)
            color = {0, 1, 0, 1};  // 緑
        else if (ratio > 0.25f)
            color = {1, 1, 0, 1};  // 黄
        else
            color = {1, 0, 0, 1};  // 赤
        view.enemyColors.push_back(color);
    }

    if (input.aim.dragging && maxDrag > 0)
    {
        float x = input.aim.anchor.x - input.aim.point.x;
        float z = input.aim.anchor.z - input.aim.point.z;
        float length = std::sqrt(x * x + z * z);
        if (std::isfinite(length) && length > 0)
        {
            if (length > maxDrag)
            {
                x = x / length * maxDrag;
                z = z / length * maxDrag;
            }
            view.arrowStart = input.playerPosition;
            view.arrowEnd.x = view.arrowStart.x + x;
            view.arrowEnd.z = view.arrowStart.z + z;
            view.showArrow = true;
        }
    }
    return view;
}
```

### 上から読む

- LimitRatioは割合を0〜1に収める。HPが5/10なら0.5。
- Build：HudInputを受けてHudViewを返す。ゲームのHPは変更しない。
- to_stringで数を文字列に変換する。負の手数は0にする。
- ボス最大HPが0なら割り算しない。
- 敵の割合を順に調べ、50%より上は緑、25%より上は黄、それ以下は赤。RGBAは各色0〜1。
- ドラッグ中のみ矢印を作る。
　仮想XZ開始座標（通常0,0）と現在の仮想座標との差を逆向きにし、
　最大ドラッグ距離を超えないようにする。
　矢印はクリック地点ではなく `input.playerPosition`（実際のプレイヤー座標）から伸ばす。
　新版でもコードは変更不要。
- 実際のDirectX描画はリードのHudRendererが行う。この担当では描画しない。
