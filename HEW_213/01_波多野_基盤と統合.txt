**書き方のルール：この資料だけで担当コードを書き終えられます。**

- 「ファイル全文」は、そのファイルを1つ完成させるコードです。共通hがすでにある場合は照合するだけで、二重に追記しません。
- 「順番に追記」は、同じcppの末尾へ番号順に書くコードです。最初のブロックから最後まで書けば完成します。途中でファイルを閉じたり別のクラスを作ったりする必要はありません。

- Visual Studioで既存のファイルがあるか確認し、ない場合は指定された物理フォルダに新規作成してプロジェクトへ登録します。日本語コメント付きの新規ファイルはUTF-8 BOM付きで保存すると安全です。

## 1. Active（GameObjectの既存機能に追加）

GameObject.hではSetActive、IsActiveとm_activeを追加。GameObject.cppではUpdate冒頭で無効ならreturn、Clone時はactiveをtrueに戻します。既に第21回相当の実装がある場合は二重に書かないこと。

`Source/Engine/Scene/GameObject.h`（ファイル全文・省略なし）

```cpp
//========================================//
// 
// GameObject.h
// 
// クラスの宣言
// public関数の宣言
// メンバ変数
// テンプレート関数
// 短いGetter / Setter
// 
//========================================//
#pragma once

#include <string>
#include <vector>
#include <memory>

#include <DirectXMath.h>

#include "Engine/Math/Transform.h"
#include "Engine/Scene/Component.h"

// すべての基本的なゲームオブジェクトの共通クラス
class GameObject
{
public:
	// 名前を受け取って初期化
	explicit GameObject(const std::string& name)
		: m_name(name)
		, m_color(1.0f, 1.0f, 1.0f, 1.0f)
	{
	}

	// ・所有者のアドレスを変えない。複製はClone()を使用する。
	// 　　GameObjectはComponentを所有しており、
	// 　　Componentは所有者のアドレスを覚えている。
	// 　　そのため、GameObjectそのものを直接コピーしたり、
	// 　　別のアドレスに移動したりすると、所有者のポインタに
	// 　　問題が発生する可能性がある。
	// 
	// 　　ただし、objects.push_back(move(pObject));は使える。
	// 　　これはGameObjectそのものを移動しているわけではなく、
	// 　　GameObjectを所有しているunique_ptrを移動しているだけだから。
	GameObject(const GameObject&) = delete;
	GameObject& operator = (const GameObject&) = delete;
	GameObject(GameObject&&) = delete;
	GameObject& operator = (GameObject&&) = delete;

	// 継承先を安全に破棄するためにvirtualを付ける
	virtual ~GameObject() = default;

	// 共通の更新処理
	// 更新は派生クラスで実装しない
	virtual void Update(float deltaTime) final;

	// 描画は派生クラスで実装する
	virtual void Draw() = 0;

	// 自分自身を複製する
	virtual std::unique_ptr<GameObject> Clone() const = 0;

	// オブジェクト名を取得
	const std::string& GetName() const
	{
		return m_name;
	}

	// GameObjectの名前を変更
	void SetName(const std::string& name)
	{
		m_name = name;
	}

	// Transformを取得
	Transform& GetTransform()
	{
		return m_transform;
	}

	// Transformの初期状態を保存
	void SaveInitialTransform()
	{
		m_initialTransform = m_transform;
	}

	// 保存した初期状態に戻す
	void ResetTransform()
	{
		m_transform = m_initialTransform;
	}

	// 実行時の有効状態（JSONには保存しない）
	void SetActive(bool active) { m_active = active; }
	bool IsActive() const { return m_active; }

	// 色を設定
	void SetColor(const DirectX::XMFLOAT4& color)
	{
		m_color = color;
	}

	// 現在の色を取得
	const DirectX::XMFLOAT4& GetColor() const
	{
		return m_color;
	}

	// コンポーネントを追加する
	bool AddComponent(std::unique_ptr<Component> pComponent);

	// 指定した種類のコンポーネントが存在するか確認
	// ・template は型を後から指定できるようする仕組み
	// 　pObject->HasComponent<AutoRotateComponent>();みたいに書くと
	// 　T には AutoRotateComponentが入る。
	template <typename T>
	T* GetComponent()
	{
		// 全コンポーネントを調べる
		for (const auto& pComponent : m_components)
		{
			// 指定した型のコンポーネントか確認
			if (auto* p = dynamic_cast<T*>(pComponent.get())) return p;
		}

		return nullptr;
	}

	// constオブジェクトから読み取り専用のComponentを取得する用
	template <typename T>
	const T* GetComponent() const
	{
		// 全コンポーネントを調べる
		for (const auto& pComponent : m_components)
		{
			// 指定した型のコンポーネントか確認
			if (auto* p = dynamic_cast<T*>(pComponent.get())) return p;
		}

		return nullptr;
	}

	template <typename T>
	bool HasComponent() const
	{
		return GetComponent<T>() != nullptr;
	}

	// RegistryやEditor向け：保存用IDで存在確認する
	bool HasComponentId(const std::string& typeId) const;

	// コンポーネントを削除する
	bool RemoveComponent(Component* pTarget);
	
	//全コンポーネントを取り除く
	void ClearComponents();

	// 取り付けられたコンポーネント一覧を取得
	const std::vector<std::unique_ptr<Component>>& GetComponents() const
	{
		return m_components;
	}

	// 読み取り専用
	const Transform& GetTransform() const
	{
		return m_transform;
	}

protected:
	// GameObjectの共通情報を複製先へコピー
	void CopyCommonStateTo(GameObject& destination) const;

private:
	std::string m_name;
	bool m_active = true;

	Transform m_transform;
	Transform m_initialTransform;

	// GameObjectがコンポーネントを所有
	std::vector<std::unique_ptr<Component>> m_components;

	// オブジェクトごとの色（RGBA）
	DirectX::XMFLOAT4 m_color;
};
```

`Source/Engine/Scene/GameObject.cpp`（ファイル全文・省略なし）

```cpp
//========================================//
// 
// GameObject.cpp
// 
//########################################
// コンポーネント更新
// 追加・削除処理
// 重複チェック
// 複数処理
// 詳細な内部実装
//########################################
// 
//========================================//

#include "GameObject.h"

#include <utility>
#include <typeinfo>

void GameObject::Update(float deltaTime)
{
	if (!m_active) return;
	for (auto& pComponent : m_components)
	{
		// 有効なコンポーネントだけ更新
		if (pComponent && pComponent->IsEnabled())
		{
			pComponent->Update(deltaTime);
		}
	}
}

bool GameObject::AddComponent(std::unique_ptr<Component> pComponent)
{
	// nullptrや、別のGameObjectを所有者にしたComponentは拒否
	// 　Enemyに取り付けたComponentがPlayerの座標を変更してしまう
	// 　みたいな間違いを防げる。
	if (!pComponent || pComponent->GetOwner() != this) return false;

	const char* id = pComponent->GetTypeId();

	if (!id || !*id || HasComponentId(id)) return false;

	// 既存のコンポーネントと型を比較
	for (const auto& pExisting : m_components)
	{
		if (!pExisting) continue;

		// 同じ型なら追加しない
		if (typeid(*pExisting) == typeid(*pComponent))
		{
			return false;
		}
	}

	// 重複していなければ追加する
	m_components.push_back(std::move(pComponent));

	return true;
}

bool GameObject::HasComponentId(const std::string& typeId) const
{
	for (const auto& pComponent : m_components)
	{
		if (pComponent && pComponent->GetTypeId() &&
			typeId == pComponent->GetTypeId()) return true;
	}

	return false;
}

bool GameObject::RemoveComponent(Component* pTarget)
{
	if (pTarget == nullptr) return false;

	// コンポーネントを先頭から調べる
	// ・it はイテレータ：vectorの要素を順番に調べるため
	// 　begin() : 最初の要素
	// 　end() 　: 最後の要素の1つ後ろの位置
	// 杉浦先生の言っていた内容は、
	// 複数の要素を削除したい時に書く
	for (auto it = m_components.begin();
		it != m_components.end(); ++it)
	{
		// 削除対象を見つけたら
		if (it->get() == pTarget)
		{
			// vectorから削除
			m_components.erase(it);

			// 削除したらループから抜けるため
			// 一つ飛ばしにならず、エラーにならない
			return true;
		}
	}

	return false;
}

void GameObject::ClearComponents()
{
	m_components.clear();
}

void GameObject::CopyCommonStateTo(GameObject& destination) const
{
	// 名前をコピー
	destination.m_name = m_name;
	destination.m_active = true;

	// Transformをコピー
	destination.m_transform = m_transform;

	// Colorをコピー
	destination.m_color = m_color;

	// 複製先のコンポーネントを一度空にする
	destination.m_components.clear();

	// 元のコンポ―ネントを順番に複製
	for (const auto& pComponent : m_components)
	{
		if (!pComponent) continue;

		// 新しい所有者を渡して複製
		std::unique_ptr<Component> pCopy =
			pComponent->Clone(&destination);

		if (pCopy)
		{
			destination.AddComponent(std::move(pCopy));
		}
	}

	// 複製時の状態を初期状態として保存
	destination.SaveInitialTransform();
}
```

## 2. BoardPointer（相対マウス入力・仮想ドラッグ）

第20回の基盤を再利用します。既に実装している場合は照合のみ。hを先に作成してからcppを作成します。

`Source/Game/Input/BoardPointer.h`（ファイル全文・省略なし）

```cpp
#include <Windows.h>
#include "Game/Systems/GameTypes.h"

class Camera;

class BoardPointer
{
public:
    // Startup.cppから呼ぶRaw Inputの入口
    static bool RegisterMouse(HWND hWnd);
    static void ReceiveRawInput(LPARAM lParam);
    static void CancelRawInput();

    // 毎フレーム1回呼ぶ（発射可能なPlay中のみ）
    PointerFrame BuildFrame(Camera& camera);
    void Reset();

private:
    bool m_bDragging = false;
    bool m_bWaitForRelease = true;
    BoardVec2 m_point;
    BoardVec2 m_right;

```

`Source/Game/Input/BoardPointer.cpp`（ファイル全文・省略なし）

```cpp
#include "BoardPointer.h"
#include "Engine/Camera/Camera.h"
#include <cmath>
#ifdef _DEBUG
#include "Engine/Debug/DebugEditor.h"
#endif

// WndProcとBuildFrameは同じメインスレッドで実行される。
// マウス1台分の移動量を次のUpdateまで貯める。
namespace
{
    HWND g_hInputWindow = nullptr;
    float g_dx = 0.0f;
    float g_dy = 0.0f;
    bool g_collecting = false;
    bool g_released = false;
    bool g_cancelled = false;
    const float kMouseSensitivity = 0.01f;
}

bool BoardPointer::RegisterMouse(HWND hWnd)
{
    if (!hWnd)
        return false;

    RAWINPUTDEVICE device{};
    device.usUsagePage = 0x01;
    device.usUsage = 0x02; // マウス
    device.dwFlags = 0;   // 通常のマウス入力も残す
    device.hwndTarget = hWnd;

    if (!RegisterRawInputDevices(&device, 1, sizeof(device)))
        return false;

    g_hInputWindow = hWnd;
    return true;
}

void BoardPointer::ReceiveRawInput(LPARAM lParam)
{
    if (!g_collecting)
        return;

    RAWINPUT input{};
    UINT size = sizeof(input);
    UINT result = GetRawInputData(
        reinterpret_cast<HRAWINPUT>(lParam), RID_INPUT,
        &input, &size, sizeof(RAWINPUTHEADER));

    if (result == static_cast<UINT>(-1) ||
        result < sizeof(RAWINPUTHEADER) + sizeof(RAWMOUSE) ||
        input.header.dwType != RIM_TYPEMOUSE)
        return;

    // 絶対座標型の機器の数値は移動量として扱わない。
    const RAWMOUSE& mouse = input.data.mouse;
    if (!(mouse.usFlags & MOUSE_MOVE_ABSOLUTE))
    {
        g_dx += static_cast<float>(mouse.lLastX);
        g_dy += static_cast<float>(mouse.lLastY);
    }

    // ボタンを離した後の移動は次の発射に混ぜない。
    if (mouse.usButtonFlags & RI_MOUSE_LEFT_BUTTON_UP)
    {
        g_released = true;
        g_collecting = false;
    }
}

void BoardPointer::CancelRawInput()
{
    g_collecting = false;
    g_released = false;
    g_dx = 0.0f;
    g_dy = 0.0f;
    g_cancelled = true;
}

void BoardPointer::Reset()
{
    m_bDragging = false;
    m_bWaitForRelease = true;
    m_point = {};
    m_right = {};
    CancelRawInput();
    g_cancelled = false;
}

PointerFrame BoardPointer::BuildFrame(Camera& camera)
{
    PointerFrame frame{};
    HWND hWnd = g_hInputWindow;

    // フォーカス喪失・最小化・サイズ変更などは発射せず中断。
    if (g_cancelled || !hWnd || GetForegroundWindow() != hWnd || IsIconic(hWnd))
    {
        Reset();
        return frame;
    }

    bool held = (GetAsyncKeyState(VK_LBUTTON) & 0x8000) != 0;
    if (m_bWaitForRelease)
    {
        if (held)
            return frame;
        m_bWaitForRelease = false;
    }

    // すでにゲーム側で開始したドラッグは画面内外を問わず続ける。
    if (m_bDragging)
    {
        float right = g_dx * kMouseSensitivity;
        float forward = -g_dy * kMouseSensitivity;
        g_dx = 0.0f;
        g_dy = 0.0f;

        // 画面の右・上方向をXZ平面の方向に変換する。
        m_point.x += right * m_right.x - forward * m_right.z;
        m_point.z += right * m_right.z + forward * m_right.x;

        frame.valid = true;
        frame.point = m_point;
        if (g_released || !held)
        {
            frame.released = true;
            Reset();
        }
        else
        {
            frame.held = true;
        }
        return frame;
    }

    // ここはドラッグ開始前だけ。UIや画面外からは開始しない。
#ifdef _DEBUG
    if (DebugEditor::WantsMouse())
    {
        Reset();
        return frame;
    }
#endif
    POINT mouse{};
    RECT client{};
    if (!GetCursorPos(&mouse) || !ScreenToClient(hWnd, &mouse) ||
        !GetClientRect(hWnd, &client) || !PtInRect(&client, mouse))
    {
        Reset();
        return frame;
    }

    frame.valid = true;
    if (!held)
        return frame;

    // 転置前のView行列から、カメラの右方向(XZ)を取得する。
    DirectX::XMFLOAT4X4 view = camera.GetViewMatrix(false);
    float length = std::sqrt(view._11 * view._11 + view._31 * view._31);
    if (!std::isfinite(length) || length < 0.0001f)
    {
        Reset();
        return PointerFrame{};
    }
    m_right.x = view._11 / length;
    m_right.z = view._31 / length;

    // クリック位置に関係なく、仮想座標(0,0)から開始する。
    m_point = {};
    m_bDragging = true;
    g_dx = 0.0f;
    g_dy = 0.0f;
    g_released = false;
    g_collecting = true;
    frame.pressed = true;
    frame.held = true;
    frame.point = m_point;
    return frame;
}
```

### 上から読む

- `RegisterMouse`：Windowsに通常のマウス入力に加えてRaw Inputを受け取るよう登録する。`Startup.cpp` から1回呼ぶ。
- `ReceiveRawInput`：`WM_INPUT` が来るたびに `lLastX/lLastY`（相対移動量）を `g_dx/g_dy` に足す。
　カーソル座標が端で止まっても移動量は取得できる。絶対座標型のデバイスは相対量として扱わない。
　左ボタンを離した通知後は移動を足さない。
- `Reset`：仮想座標と進行中のドラッグを初期化。一度ボタンを離すまで新規開始しない。
- `BuildFrame`：1フレーム1回だけ。開始前にフォーカス・画面内・ImGuiを確認。
　押した位置は問わず仮想座標を(0,0)にする。開始後は相対移動量を積算する。
　カメラの右方向と画面上方向をXZへ変換。
　`kMouseSensitivity=0.01f` は1カウント当たりのワールド単位（最大ドラッグ3なら約300カウント）。
　離したフレームは現在の仮想座標を返してからResetする。
- ここでプレイヤーの中心や半径を使用しない。発射方向・速度は栗本のクラスが計算する。カーソルを強制移動・非表示にしない。

### 穴埋め（練習のみ）

`g_dx += 【空欄】;` 生のX移動量を足す。正解：`static_cast<float>(mouse.lLastX)`。

`Source/App/Startup.cpp`（ファイル全文・省略なし）

```cpp
#include <windows.h>

#include "Defines.h"
#include "Main.h"
#include "Game/Input/BoardPointer.h"

#include <stdio.h>
#include <crtdbg.h>

#ifdef _DEBUG
#include "Engine/Debug/DebugEditor.h"
#endif

// timeGetTime周りの使用
#pragma comment(lib, "winmm.lib")

namespace
{
	bool g_resizePending = false;

	UINT g_pendingWidth = SCREEN_WIDTH;
	UINT g_pendingHeight = SCREEN_HEIGHT;
}

//--- プロトタイプ宣言
LRESULT CALLBACK WndProc(HWND hWnd, UINT message, WPARAM wParam, LPARAM lParam);

// エントリポイント
int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPSTR lpCmdLine, int nCmdShow)
{
	// ウィンドウ作成前にDPI対応を有効化
	SetProcessDpiAwarenessContext(
		DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE_V2
	);

	_CrtSetDbgFlag(_CRTDBG_ALLOC_MEM_DF | _CRTDBG_LEAK_CHECK_DF);

	//--- 変数宣言
	WNDCLASSEX wcex;
	HWND hWnd;
	MSG message;

	// ウィンドクラス情報の設定
	ZeroMemory(&wcex, sizeof(wcex));
	wcex.hInstance = hInstance;
	wcex.lpszClassName = "Class Name";
	wcex.lpfnWndProc = WndProc;
	wcex.style = CS_CLASSDC | CS_DBLCLKS;
	wcex.cbSize = sizeof(WNDCLASSEX);
	wcex.hIcon = LoadIcon(NULL, IDI_APPLICATION);
	wcex.hIconSm = wcex.hIcon;
	wcex.hCursor = LoadCursor(NULL, IDC_ARROW);
	wcex.hbrBackground = (HBRUSH)GetStockObject(WHITE_BRUSH);

	// ウィンドウクラス情報の登録
	if (!RegisterClassEx(&wcex))
	{
		MessageBox(NULL, "Failed to RegisterClassEx", "Error", MB_OK);
		return 0;
	}

	// ウィンドウの作成
	RECT rect = { 0, 0, SCREEN_WIDTH, SCREEN_HEIGHT };
	DWORD style = WS_OVERLAPPEDWINDOW;
	DWORD exStyle = 0;
	AdjustWindowRectEx(&rect, style, FALSE, exStyle);
	hWnd = CreateWindowEx(
		exStyle, wcex.lpszClassName,
		APP_TITLE, style,
		CW_USEDEFAULT, CW_USEDEFAULT,
		rect.right - rect.left, rect.bottom - rect.top,
		HWND_DESKTOP,
		NULL, hInstance, NULL
	);

	// Raw Inputでマウスの移動量を受け取る（Debug/Release共通）
	if (!BoardPointer::RegisterMouse(hWnd))
	{
		MessageBoxA(hWnd, "Raw mouse registration failed.", "Input Error", MB_OK | MB_ICONERROR);
		if (hWnd)
			DestroyWindow(hWnd);
		UnregisterClass(wcex.lpszClassName, hInstance);
		return 0;
	}

	// ウィンドウの表示
	ShowWindow(hWnd, nCmdShow);
	UpdateWindow(hWnd);

	// 初期化処理
	if (FAILED(Init(hWnd, SCREEN_WIDTH, SCREEN_HEIGHT)))
	{
		Uninit();
		UnregisterClass(wcex.lpszClassName, hInstance);
		return 0;
	}

	// 初期化完了時の実際のクライアント領域を確認
	RECT clientRect = {};

	GetClientRect(hWnd, &clientRect);

	g_pendingWidth =
		clientRect.right - clientRect.left;
	g_pendingHeight =
		clientRect.bottom - clientRect.top;

	g_resizePending = true;

	//--- FPS制御
	timeBeginPeriod(1);
	DWORD countStartTime = timeGetTime();
	DWORD preExecTime = countStartTime;
#ifdef _DEBUG
	// 高精度タイマーの周波数
	LARGE_INTEGER performanceFrequency = {};
	QueryPerformanceFrequency(&performanceFrequency);

	// FPS計測を開始した時刻
	LARGE_INTEGER fpsStartTime = {};
	QueryPerformanceCounter(&fpsStartTime);

	// 計測期間内に描画したフレーム数
	unsigned int measuredFrames = 0;

	// Update + Drawにかかった合計時間
	double totalWorkMs = 0.0f;
#endif

	//--- ウィンドウの管理
	while (1)
	{
		if (PeekMessage(&message, NULL, 0, 0, PM_NOREMOVE))
		{
			if (!GetMessage(&message, NULL, 0, 0))
			{
				break;
			}
			else
			{
				TranslateMessage(&message);
				DispatchMessage(&message);
			}
		}
		else
		{
			if (IsIconic(hWnd))
			{
				WaitMessage();
				continue;
			}

			if (g_resizePending)
			{
				const UINT newWidth = g_pendingWidth;
				const UINT newHeight = g_pendingHeight;
				
				g_resizePending = false;

				HRESULT hr = ResizeGame(newWidth, newHeight);

				if (FAILED(hr))
				{
					char errorText[256];

					sprintf_s(
						errorText,
						"Failed to resize DirectX.\nHRESULT: 0x%08X",
						static_cast<unsigned int>(hr)
					);

					MessageBoxA(
						hWnd,
						errorText,
						"Resize Error",
						MB_OK | MB_ICONERROR
					);

					DestroyWindow(hWnd);
					continue;
				}
			}

			DWORD nowTime = timeGetTime();
			float diff = static_cast<float>(nowTime - preExecTime);
			if (diff >= 1000.0f / fFPS)
			{
#ifdef _DEBUG
				//========================================
				// フレーム処理開始時刻
				//========================================
				LARGE_INTEGER frameStart = {};
				QueryPerformanceCounter(&frameStart);

#endif
				//========================================
				// ゲームを更新・描画
				//========================================
				Update();
				Draw();

#ifdef _DEBUG
				//========================================
				// フレーム処理終了時刻
				//========================================
				LARGE_INTEGER frameEnd = {};
				QueryPerformanceCounter(&frameEnd);

				const double frequency =
					static_cast<double>(
						performanceFrequency.QuadPart
						);

				// 今回のUpdate + Drawの所要時間
				const double workMs =
					static_cast<double>(
						frameEnd.QuadPart -
						frameStart.QuadPart
						) * 1000.0 / frequency;

				totalWorkMs += workMs;

				++measuredFrames;

				//========================================
				// FPS計測期間の長さ
				//========================================
				const double elapsedSeconds =
					static_cast<double>(
						frameEnd.QuadPart -
						fpsStartTime.QuadPart
						) / frequency;

				//========================================
				// 約1秒ごとにタイトルを更新
				//========================================
				if (elapsedSeconds >= 1.0)
				{
					// 実際に描画されたFPS
					const double actualFPS =
						static_cast<double>(
							measuredFrames
							) / elapsedSeconds;

					// 平均処理時間
					const double averageWorkMs =
						totalWorkMs /
						static_cast<double>(
							measuredFrames
							);

					// タイトル文字列を作成
					char windowTitle[160] = {};

					sprintf_s(
						windowTitle,
						sizeof(windowTitle),
						"%s | FPS: %.1f | Work: %.2f ms",
						APP_TITLE,
						actualFPS,
						averageWorkMs
					);

					// Windowsのタイトルバーを更新
					SetWindowTextA(
						hWnd,
						windowTitle
					);

					// 次の計測期間を開始
					measuredFrames = 0;
					totalWorkMs = 0.0;
					fpsStartTime = frameEnd;
				}
#endif

				preExecTime = nowTime;
			}
		}
	}


	// 終了時
	timeEndPeriod(1);
	Uninit();
	UnregisterClass(wcex.lpszClassName, hInstance);

	return 0;
}

// ウィンドウプロシージャ
LRESULT CALLBACK WndProc(HWND hWnd, UINT message, WPARAM wParam, LPARAM lParam)
{
	// ImGuiより先にRaw Inputを受け取り、Windowsの既定処理へ渡す。
	if (message == WM_INPUT)
	{
		BoardPointer::ReceiveRawInput(lParam);
		return DefWindowProc(hWnd, message, wParam, lParam);
	}

	if (message == WM_KILLFOCUS || message == WM_CANCELMODE ||
		message == WM_ENTERSIZEMOVE ||
		(message == WM_ACTIVATEAPP && !wParam) ||
		(message == WM_SIZE && wParam == SIZE_MINIMIZED))
	{
		BoardPointer::CancelRawInput();
	}

	if (message == WM_SIZE)
	{
		if (wParam != SIZE_MINIMIZED)
		{
			UINT width = LOWORD(lParam);
			UINT height = HIWORD(lParam);

			if (width > 0 && height > 0)
			{
				g_pendingWidth = width;
				g_pendingHeight = height;
				g_resizePending = true;
			}
		}
	}

#ifdef _DEBUG
	if (DebugEditor::HandleWndProc(
		hWnd, message, wParam, lParam))
	{
		return true;
	}
#endif

	switch (message)
	{
	case WM_GETMINMAXINFO:
	{
		MINMAXINFO* pInfo =
			reinterpret_cast<MINMAXINFO*>(lParam);

		pInfo->ptMinTrackSize.x = 900;
		pInfo->ptMinTrackSize.y = 600;

		return 0;
	}

	case WM_DESTROY:
		PostQuitMessage(0);
		return 0;

#ifdef _DEBUG
	case WM_KEYDOWN:
		if (wParam == VK_ESCAPE)
		{
			DestroyWindow(hWnd);
			return 0;
		}
		break;
#endif
	}

	return DefWindowProc(hWnd, message, wParam, lParam);
}
```

`WM_INPUT` を処理した後は `DefWindowProc` を呼ぶ。

```cpp
void HudRenderer::Text(float x, float y, float p, const std::string &text, HudColor c)
{
    for (char ch : text)
    {
        const auto *rows = Glyph(ch);
        if (rows)
            for (int r = 0; r < 7; ++r)
                for (int k = 0; k < 5; ++k)
                    if (rows[r] & (1u << (4 - k)))

{
    if (!m_ready || !GetRenderWidth() || !GetRenderHeight())
        return;
    const float w = float(GetRenderWidth()), h = float(GetRenderHeight());
    const float u = std::max(1.0f, std::min(w / 640.0f, h / 360.0f));

    GetContext()->RSGetState(&pRaster);

    Sprite::SetView(identity);
    Sprite::SetProjection(projection);
    Sprite::SetVertexShader(nullptr);
    Sprite::SetPixelShader(nullptr);
    Sprite::SetTexture(&m_white);
    Sprite::SetUVPos(XMFLOAT2(0, 0));
    Sprite::SetUVScale(XMFLOAT2(1, 1));
    if (hasBoss)
    {
        Rect(12 * u, 12 * u, 210 * u, 42 * u, bg);
        Text(18 * u, 17 * u, 2 * u, "BOSS HP", white);
        Rect(18 * u, 34 * u, 198 * u, 13 * u, {0.25f, 0.25f, 0.25f, 1});

}
```

```cpp
#include "Engine/Scene/Component.h"

class ComponentRegistry

		// JSON読込関数
```

```cpp
// JSONから3つのfloat値を読み込み
void ReadVector3(const json &value, float result[3], const char *field) {
  if (!value.is_array() || value.size() != 3) {
    throw runtime_error(string(field) + "must contain 3 numbers.");
  }

  for (int i = 0; i < 3; ++i) {
    result[i] = value.at(i).get<float>();
  }
}
} // namespace

const vector<ComponentRegistry::Entry> &ComponentRegistry::GetEntries() {
  // ・ラムダ式で記述
  // 　　別の名前付き関数を定義しなくても
  // 　　その場で関数の処理を記述できる記法
  static const vector<Entry> entries = {
      //========================================
      // AutoRotate
      //========================================
      {"AutoRotate",  // JSON保存用のID
       "Auto Rotate", // Inspectorの表示名

       // 生成
       [](GameObject *owner) -> unique_ptr<Component> {
         return unique_ptr<Component>(
             new AutoRotateComponent(owner, 0.0f, 360.0f, 0.0f));
       },

       // 保存
       [](const Component &base, json &data) {
         // 今回の場合、
         // 〇dynamic_cast … 実際に指定した派生型へ
         // 　　　　　　　　　変換できるか確認する。
         // ×static_cast  … 基底クラスからの変換では、
         // 　　　　　　　　　実際の派生型を実行時に確認しない。
         // これは登録ミスによって別のComponentが渡された場合、
         // 安全に失敗するとは限らないため。
         const AutoRotateComponent &c =
             dynamic_cast<const AutoRotateComponent &>(base);

         const float *speed = c.GetRotationSpeed();

         data["speed"] = {speed[0], speed[1], speed[2]};
       },

       // 読み込み
       [](Component &base, const json &data) {
         float speed[3];

         ReadVector3(data.at("speed"), speed, "speed");

         AutoRotateComponent &c = dynamic_cast<AutoRotateComponent &>(base);

         c.SetRotationSpeed(speed[0], speed[1], speed[2]);
       }},

      //========================================
      // Move
      //========================================
      {"Move", "Move",

       [](GameObject *owner) -> unique_ptr<Component> {
         return unique_ptr<Component>(
             new MoveComponent(owner, 1.0f, 1.0f, 0.0f));
       },

       // 保存
       [](const Component &base, json &data) {
         const MoveComponent &c = dynamic_cast<const MoveComponent &>(base);

         const float *velocity = c.GetVelocity();

         data["velocity"] = {velocity[0], velocity[1], velocity[2]};
       },

       // 読み込み
       [](Component &base, const json &data) {
         float velocity[3];

         ReadVector3(data.at("velocity"), velocity, "velocity");

         MoveComponent &c = dynamic_cast<MoveComponent &>(base);

         c.SetVelocity(velocity[0], velocity[1], velocity[2]);
       }}

      //========================================
      // 追加はここから
      //========================================
      ,
      {"BallMotion", "BallMotion",
       [](GameObject *pOwner) -> unique_ptr<Component> {
         return std::make_unique<BallMotionComponent>(pOwner);
       },
       [](const Component &base, json &data) {

         data["deceleration"] = c.GetDeceleration();
         data["stopSpeed"] = c.GetStopSpeed();

         c.Configure(data.at("deceleration").get<float>(),

       }},
      {"PlayerLaunch", "PlayerLaunch",

         return std::make_unique<PlayerLaunchComponent>(pOwner);
       },
       [](const Component &base, json &data) {
         const auto &c = dynamic_cast<const PlayerLaunchComponent &>(base);
         data["minDrag"] = c.GetMinDrag();
         data["maxDrag"] = c.GetMaxDrag();
         data["maxSpeed"] = c.GetMaxSpeed();
       },
       [](Component &base, const json &data) {
         auto &c = dynamic_cast<PlayerLaunchComponent &>(base);
         c.Configure(data.at("minDrag").get<float>(),
                     data.at("maxDrag").get<float>(),
                     data.at("maxSpeed").get<float>());
       }},
      {"PlayerInput", "PlayerInput",
       [](GameObject *pOwner) -> unique_ptr<Component> {
         return std::make_unique<PlayerInputComponent>(pOwner);
       },
       [](const Component &base, json &data) {
         (void)base;
         (void)data;
       },
       [](Component &base, const json &data) {
         (void)base;
         (void)data;
       }},
      {"ReflectionCounter", "ReflectionCounter",
       [](GameObject *pOwner) -> unique_ptr<Component> {
         return std::make_unique<ReflectionCounterComponent>(pOwner);
       },
       [](const Component &base, json &data) {
         (void)base;
         (void)data;
       },
       [](Component &base, const json &data) {
         (void)base;
         (void)data;
       }},
      {"Health", "Health",
       [](GameObject *pOwner) -> unique_ptr<Component> {
         return std::make_unique<HealthComponent>(pOwner);
       },
       [](const Component &base, json &data) {
         const auto &c = dynamic_cast<const HealthComponent &>(base);
         data["maxHP"] = c.GetMaxHP();
       },
       [](Component &base, const json &data) {
         auto &c = dynamic_cast<HealthComponent &>(base);
         c.Reset(data.at("maxHP").get<int>());
       }},
      {"Collider", "Collider",
       [](GameObject *pOwner) -> unique_ptr<Component> {
         return std::make_unique<ColliderComponent>(pOwner);
       },
       [](const Component &base, json &data) {
         const auto &c = dynamic_cast<const ColliderComponent &>(base);
         const auto &v = c.GetConfig();
         data["role"] = int(v.role);
         data["shape"] = int(v.shape);
         data["radius"] = v.radius;
         data["halfExtent"] = {v.halfExtent.x, v.halfExtent.z};
         data["restitution"] = v.restitution;
       },
       [](Component &base, const json &data) {
         auto &c = dynamic_cast<ColliderComponent &>(base);
         ColliderConfig v;
         v.role = GameRole(data.at("role").get<int>());
         v.shape = ColliderShape(data.at("shape").get<int>());
         v.radius = data.at("radius").get<float>();

           throw runtime_error("Invalid halfExtent");
         v.halfExtent = {h.at(0).get<float>(), h.at(1).get<float>()};
         v.restitution = data.at("restitution").get<float>();
         c.Configure(v);
       }}};

  return entries;

const ComponentRegistry::Entry *
ComponentRegistry::Find(const std::string &typeId) {
  const vector<Entry> &entries = GetEntries();

  // 登録一覧を先頭から調べる
  for (const Entry &entry : entries) {
    if (typeId == entry.typeId) {
      return &entry;
    }
  }

  // 見つからなかった場合はnullptrで返す
  return nullptr;
}
```

```cpp
void SceneGame::UpdatePlayerInput()
{
    GameObject* pPlayer = Player(m_objects);
    if (!pPlayer)
    {
        m_boardPointer.Reset();
        return;
    }
    auto* pInput = Enabled<PlayerInputComponent>(*pPlayer);
    auto* pLaunch = Enabled<PlayerLaunchComponent>(*pPlayer);
    auto* pMotion = Enabled<BallMotionComponent>(*pPlayer);
    if (!pInput || !pLaunch || !pMotion || !m_pPlayCamera || !m_turn.CanLaunch())
    {
        m_boardPointer.Reset();
        if (pInput)
            pInput->Cancel();
        return;
    }

    // プレイヤーのクリック判定は不要。移動量だけを取得する。
    PointerFrame frame = m_boardPointer.BuildFrame(*m_pPlayCamera);
    pInput->Sample(frame, true);

    // ドラッグを離した時だけ速度を作る。
    BoardVec2 v;
    if (!pLaunch->TryBuildVelocity(pInput->GetAim(), v))
        return;
    float speed = std::sqrt(v.x * v.x + v.z * v.z);
    if (speed <= pMotion->GetStopSpeed())
        return;
    if (!m_turn.TryConsumeShot())
        return;

    m_collision.Reset();
    for (auto& pObject : m_objects)
    {
        if (!pObject || !pObject->IsActive())
            continue;
        auto* pCounter = Enabled<ReflectionCounterComponent>(*pObject);
        if (pCounter)
            pCounter->Reset();
    }
    pMotion->SetVelocity(v);
    pInput->Cancel();
}
```

```cpp
void SceneGame::ApplyCollisionNotices(const vector<CollisionNotice> &notices) {
  // CollisionSystemが攻撃できる組み合わせだけを通知する。
  for (const CollisionNotice& notice : notices)
  {
    GameObject* pTarget = notice.target;
    if (!notice.attacker || !pTarget)
      continue;
    if (!notice.attacker->IsActive() || !pTarget->IsActive())
      continue;
    auto* pHP = Enabled<HealthComponent>(*pTarget);
    if (!pHP)
      continue;

    int damage = DamageFormula::Calculate(kBaseDamage, kMultiplier, notice.reflectionCount);
    pHP->ApplyDamage(damage);
    if (pHP->IsDead())
    {
      auto* pMotion = Enabled<BallMotionComponent>(*pTarget);
      if (pMotion) pMotion->Stop();
      pTarget->SetActive(false); // vectorから削除せず、無効にする。
    }
  }
}
```

```cpp
      Geometory::AddLine(tip, left, color);
```

```cpp
SceneGame::SceneGame()
    : m_pCamera(nullptr), m_pPlayCamera(nullptr), m_showDebugGrid(true) {
  //========================================
  // Groundを生成
  //========================================
  unique_ptr<Ground> pGround(new Ground());

  pGround->SetPos(0.0f, 0.0f, 0.0f);
  pGround->SetScale(3.0f, 0.4f, 3.0f);

  // 現在のTransformを初期状態として保存
  pGround->SaveInitialTransform();

  // GameObjectの配列に追加
  m_objects.push_back(move(pGround));

  //========================================
  // Goalを生成する
  //========================================
  unique_ptr<Goal> pGoal(new Goal());

  pGoal->SetPos(0.0f, 1.0f, 0.0f);
  pGoal->SetScale(0.5f, 0.5f, 0.5f);

  pGoal->SaveInitialTransform();

  m_objects.push_back(move(pGoal));

  //========================================
  // カメラ
  //========================================
  // 編集用カメラはDebugのみ
#ifdef _DEBUG
  m_pCamera.reset(new CameraDebug());
#endif

  // Play用カメラはDebug・Release共通
  m_pPlayCamera.reset(new CameraPlay());

  LoadGameplayStage();
  if (!m_hudRenderer.Initialize())
    MessageBoxA(nullptr, "HUD texture creation failed.", "HUD Error",
                MB_OK | MB_ICONERROR);
#ifndef _DEBUG
  ResetGameplay();
#endif
}

//========================================
// 更新処理
//========================================
void SceneGame::Update() {
  // 更新で進める時間
  const float deltaTime = 1.0f / FPS;

  // ゲームを更新するかどうか
  bool shouldUpdateGame = true;

#ifdef _DEBUG
  //========================================
  // F1でモード切り替え
  //========================================
  if (IsKeyTrigger(VK_F1)) {
    if (m_isEditMode) {
      // Edit -> Play
      EnterPlayMode();
    } else {
      // Play -> Edit
      ExitPlayMode();
    }
  }

  // Edit Mode中はゲームを更新しない
  shouldUpdateGame = !m_isEditMode;
  UpdateColliderDebug();
#endif

  //========================================
  // ゲームの更新処理
  //========================================
  if (shouldUpdateGame) {
    if (IsKeyTrigger('R')) {
#ifdef _DEBUG
      ExitPlayMode();
      EnterPlayMode();
#else
      if (LoadGameplayStage())
        ResetGameplay();
#endif
    }
    UpdateGameplay(deltaTime);
  }

  //========================================
  // カメラ更新
  //========================================
#ifdef _DEBUG
  if (m_isEditMode) {
    // Edit Modeでは自由操作カメラ
    m_pCamera->Update();
  } else {
    // Play Modeでは固定カメラ
    m_pPlayCamera->Update();
  }
#else
  // Releaseでは常にPlay専用カメラ
  m_pPlayCamera->Update();
#endif
}

// 描画処理
//========================================
void SceneGame::Draw() {
  //========================================
  // ライティング設定
  //========================================
  ShaderList::SetLight(XMFLOAT4(1.2f, 1.2f, 1.2f, 1.0f),
                       XMFLOAT3(0.0f, -1.0f, 0.0f));

  //========================================
  // 使用するカメラを決定
  //========================================
  // 基本的にはPlay用カメラを使用
  Camera *pRenderCamera = m_pPlayCamera.get();

#ifdef _DEBUG
  // Edit Modeの場合だけDebugカメラを使用
  if (m_isEditMode) {
    pRenderCamera = m_pCamera.get();
  }
#endif

  //========================================
  // カメラ行列の設定
  //========================================
  const XMFLOAT4X4 view = pRenderCamera->GetViewMatrix();
  const XMFLOAT4X4 projection = pRenderCamera->GetProjectionMatrix();

  Geometory::SetView(view);
  Geometory::SetProjection(projection);

  // FBXオブジェクトにもカメラ行列を設定
  FbxObject::SetCameraMatrices(view, projection);

  //========================================
  // すべてのGameOjbectを描画
  //========================================
  for (auto &pObject : m_objects) {
    if (pObject && pObject->IsActive())
      pObject->Draw();
  }

  //========================================
  // Edit Modeでのみグリッドを描画
  //========================================
#ifdef _DEBUG
  if (m_isEditMode && m_showDebugGrid) {
    DrawDebugGrid();
  }
#endif

#ifdef _DEBUG
  // モードに関係なく表示する（Debugビルド限定）。
  DrawColliderDebugPanel();
  DrawColliderDebug();
  if (m_isEditMode)
    DrawEditor();
  else
    DrawGameplayHUD();
#else
  DrawGameplayHUD();
#endif
}

//========================================
// Edit Modeのエディタ処理
//========================================
void SceneGame::DrawEditor() {
  //========================================
  // 3D画面内でのオブジェクト選択
  //========================================
  float mouseX = 0.0f;
  float mouseY = 0.0f;
  float viewportWidth = 0.0f;
  float viewportHeight = 0.0f;

  // 3D画面が左クリックされたか確認
  if (DebugEditor::TryGetSceneClick(mouseX, mouseY, viewportWidth,
                                    viewportHeight)) {
    // Raycastでオブジェクトを探す
    int selectedIndex = SceneRayPicker::Pick(
        m_objects, *m_pCamera, mouseX, mouseY, viewportWidth, viewportHeight);

    // オブジェクトに当たった場合だけ選択
    if (selectedIndex >= 0) {
      DebugEditor::SetSelectedObjectIndex(selectedIndex);
    }
  }

  //========================================
  // UIから命令を受け取る
  //========================================
  EditorCommand command = DebugEditor::DrawScene(m_objects, m_showDebugGrid);

  //========================================
  // 命令を実行する
  //========================================
  SceneEditResult result = SceneEditActions::Execute(
      command, m_objects, *m_pPlayCamera, kStageFilePath);

  //========================================
  // 実行結果を反映する
  //========================================
  if (result.selectionChanged) {
    DebugEditor::SetSelectedObjectIndex(result.selectedIndex);
  }

  //========================================
  // エラーがあれば表示
  //========================================
  if (!result.error.empty()) {
    MessageBoxA(nullptr, result.error.c_str(), "Scene Editor Error",
                MB_OK | MB_ICONERROR);
  }

  //========================================
  // Playカメラ設定
  //========================================
  DebugEditor::DrawPlayCamera(*m_pPlayCamera, *m_pCamera);
}

void SceneGame::DrawColliderDebug()
{
    if (!m_bShowColliders)
        return;

    // 深度とブレンドの元の設定を保存しておく。
    ID3D11DepthStencilState* pDepth = nullptr;
    UINT stencil = 0;
    ID3D11BlendState* pBlend = nullptr;
    FLOAT blendFactor[4] = {};
    UINT sampleMask = 0;
    GetContext()->OMGetDepthStencilState(&pDepth, &stencil);
    GetContext()->OMGetBlendState(&pBlend, blendFactor, &sampleMask);

    // モデルに隠れず、半透明で線を描く。
    SetDepthTest(false);
    SetBlendMode(BLEND_ALPHA);
    for (auto& pObject : m_objects)
    {
        if (!pObject || !pObject->IsActive())
            continue;
        auto* pCollider = pObject->GetComponent<ColliderComponent>();
        if (pCollider)
            pCollider->DrawDebug();
    }

    // 後から描くHUDなどに影響させない。
    GetContext()->OMSetDepthStencilState(pDepth, stencil);
    GetContext()->OMSetBlendState(pBlend, blendFactor, sampleMask);
    if (pDepth)
        pDepth->Release();
    if (pBlend)
        pBlend->Release();
}
#endif

void SceneGame::OnResize(unsigned int width, unsigned int height) {
  if (width == 0 || height == 0)
    return;

  m_boardPointer.Reset();

  // 新しい画面の縦横比
  const float aspect = static_cast<float>(width) / static_cast<float>(height);

  //========================================
  // Edit用カメラ
  //========================================
  if (m_pCamera) {
    m_pCamera->SetAspectRatio(aspect);
  }

  //========================================
  // Play用カメラ
  //========================================
  if (m_pPlayCamera) {
    m_pPlayCamera->SetAspectRatio(aspect);
  }
}

//========================================
// グリッド描画処理
//========================================
void SceneGame::DrawDebugGrid() {
#ifdef _DEBUG
  const int gridCount = 10;
  const float gridSize = 1.0f;
  const float extent = gridCount * gridSize;

  const XMFLOAT4 gridColor(0.5f, 0.5f, 0.5f, 1.0f);

  // ｘ方向とｚ方向にグリッドを生成
  for (int i = -gridCount; i <= gridCount; i++) {
    // 中央線はｘｙｚ軸と重なるので描画しない
    if (i == 0)
      continue;

    float offset = i * gridSize;

    // ｚ方向の線
    Geometory::AddLine(XMFLOAT3(offset, 0, -extent),
                       XMFLOAT3(offset, 0, extent), gridColor);

    // ｘ方向の線
    Geometory::AddLine(XMFLOAT3(-extent, 0, offset),
                       XMFLOAT3(extent, 0, offset), gridColor);
  }

  // ｘ軸：赤
  Geometory::AddLine(XMFLOAT3(-extent, 0, 0), XMFLOAT3(extent, 0, 0),
                     XMFLOAT4(1, 0, 0, 1));

  // ｙ軸：緑
  Geometory::AddLine(XMFLOAT3(0, 0, 0), XMFLOAT3(0, 5, 0),
                     XMFLOAT4(0, 1, 0, 1));

  // ｚ軸：青
  Geometory::AddLine(XMFLOAT3(0, 0, -extent), XMFLOAT3(0, 0, extent),
                     XMFLOAT4(0, 0, 1, 1));

  Geometory::DrawLines();
#endif
}

//========================================
// Play Modeを開始する処理
//========================================
#ifdef _DEBUG
void SceneGame::EnterPlayMode() {
  // すでにPlay Modeなら何もしない
  if (!m_isEditMode)
    return;

  //========================================
  // Play用のオブジェクトを作る
  //========================================
  vector<unique_ptr<GameObject>> playObjects;

  // vectorにどれぐらい要素を入れるかが分かっている場合、
  // reserve で、あらかじめ必要な容量を確保する
  playObjects.reserve(m_objects.size());

  for (const auto &pObject : m_objects) {
    if (!pObject) {
      MessageBoxA(nullptr, "Null GameObject was found.", "Play Mode Error",
                  MB_OK | MB_ICONERROR);
      return;
    }

    // Cloneで別のオブジェクトを生成
    unique_ptr<GameObject> pClone = pObject->Clone();

    // 複製失敗時
    if (!pClone) {
      MessageBoxA(nullptr, "Failed to clone GameObject.", "Play Mode Error",
                  MB_OK | MB_ICONERROR);
      return;
    }

    playObjects.push_back(move(pClone));
  }

  //========================================
  // 元の編集オブジェクトを保管
  //========================================
  m_editObjects.swap(m_objects);

  //========================================
  // Play用の複製へ切り替える
  //========================================
  m_objects.swap(playObjects);

  // Play Modeへ変更
  m_isEditMode = false;
  ResetGameplay();
}

//========================================
// Play Modeを終了する処理
//========================================
void SceneGame::ExitPlayMode() {
  // すでにExit Modeなら何もしない
  if (m_isEditMode)
    return;

  //========================================
  // 編集用オブジェクトを復元
  //========================================
  m_objects.swap(m_editObjects);

  //========================================
  // Play用のオブジェクトを破棄
  //========================================
  m_boardPointer.Reset();
  m_collision.Reset();
  m_editObjects.clear();
  m_gameReady = false;

  // Edit Modeへ変更
  m_isEditMode = true;
}
#endif
```

## 補足

- Enabled<T> は「Componentを探し、有効か確認する」補助関数。テンプレートは今までのGameObject::GetComponentと同様。
- `unique_ptr` 管理は既存エンジンに従う。moveで所有権をvectorに移す。
- `SceneGame.cpp` のモード切替・Editor部分を移植し直す必要はない。第15章に掲載した正本を参照し、ゲーム部分とのつなぎのみ採用する。
- DirectXの描画処理・BoardPointerは配布する基盤。無理に初学者向けに行列や描画状態の処理を消さない。
